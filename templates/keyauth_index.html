# KRISHU X AUTH — PYTHON SDK
# ═════════════════════════════════════════════════════════════════════════════
import requests, hashlib, platform

SERVER_URL = "http://127.0.0.1:5000"
APP_ID     = "app_krishu_main"

def get_hwid():
    raw = platform.node() + "-" + platform.machine()
    return hashlib.sha256(raw.encode()).hexdigest()

def authenticate_license(license_key):
    # Handshake init
    init_res = requests.post(f"{SERVER_URL}/api/v1/init", json={"appId": APP_ID}).json()
    if not init_res.get("success"):
        return False, init_res.get("message")

    # 1-Click License Login
    res = requests.post(f"{SERVER_URL}/api/v1/license", json={
        "appId": APP_ID,
        "key": license_key,
        "hwid": get_hwid()
    }).json()

    return res.get("success"), res.get("message")

if __name__ == "__main__":
    key = input("Enter Krishu License Key: ").strip()
    ok, msg = authenticate_license(key)
    print(f"Result: {msg}")
